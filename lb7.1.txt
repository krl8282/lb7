#include <iostream>
#include <iomanip>
#include <math.h>
using namespace std;
int main()
{
    double y, x;
    double h;
    x = 3.0;
    h = 0.1;
    while (x <= 5)
    {
        y = pow(x, 1.0 / 2.0) + cos(x) - 3;
        cout << setw(10) << "x=" << fixed << setprecision(1) << x << setw(10) << "y=" << fixed << setprecision(4) << y << endl;
        x += h;
    }
}
