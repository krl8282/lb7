#include <iostream>
#define _USE_MATH_DEFINES
#include <math.h>
#include <windows.h>
#include <iomanip>
using namespace std;
double func_A (float z, float x, float y) { return log(z + pow(x, 2)) + pow(sin(x / y), 2) - cos(20 * M_PI / 180); }
double func_B (float x, float z, float y, float c, float a) { return exp(-1 * c * z) * ((x + sqrt(x + a)) / (x - sqrt(fabs(x - y)))); }

int main()
{
    float x, y, z, c, a;
    SetConsoleCP(1251);
    SetConsoleOutputCP(1251);
    cout << "Введіть x , y , z , c , a" << endl;
    cin >> x >> y >> z >> c >> a;
    if (z + pow(x, 2) > 0 && (x + a) > 0 && (x - sqrt(fabs(x - y))) != 0 && y!=0) {
        double A = func_A(z, x, y);
        double B = func_B(x, z, y, c, a);
        int intB = static_cast<int> (B);
        if (intB != 0) {
            int C = static_cast<int>(A) / intB;
            cout << "A= " << A << "\t" << "B= " << B << "\t" << "C=" << C;
        }
        else cout << "B=0,тому ділення С=А/В неможливе.";
    }
    else cout << "Помилка введених данних.";
}
